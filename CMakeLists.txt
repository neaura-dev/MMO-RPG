cmake_minimum_required(VERSION 3.20)

project(MMO-RPG VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE MMO_RPG_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

add_executable(MMO-RPG ${MMO_RPG_SOURCES})
target_include_directories(MMO-RPG PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/domain")
